Coco & Drift - Minimal Clean One-Page Site with Simple Admin
Files: this single HTML file contains everything (HTML + CSS + JS).
How it works:
 - Public users see a clean white product grid. Each product has a "Buy on Redbubble" button that opens the product URL.
 - Admin: open the "Admin" button at top-right. Log in with the default password (change it in the script below).
 - Admin panel (client-side) lets you add/edit/delete products. Products are saved to browser localStorage, so only you on this browser/device will see and manage them.
 - To make the admin exclusive to you across devices, host this file on a server with HTTPS and keep the password secret, or replace client-side admin with a small backend later.

Important: Replace placeholder Redbubble links and image URLs with your real product links/images.

To publish:
 1) Save this file as index.html
 2) Upload to 000webhost, Hostinger, or to a GitHub repo and enable GitHub Pages.
 3) If using GitHub Pages, the product images should be hosted publicly (you can push them to the repo or use image hosting).

Change default admin password:
 - In the script near the bottom, change DEFAULT_ADMIN_PASSWORD to your chosen password.

*/-->

  <script>
    // ====== CONFIG ======
    // Change this to your chosen admin password before you publish.
    const DEFAULT_ADMIN_PASSWORD = 'secret123'; // <-- change this now

    // ====== App ======
    const productKey = 'coco_drift_products_v1';
    const adminModal = document.getElementById('adminModal');
    const adminBtn = document.getElementById('adminBtn');
    const closeAdmin = document.getElementById('closeAdmin');
    const loginBox = document.getElementById('loginBox');
    const editor = document.getElementById('editor');
    const loginBtn = document.getElementById('loginBtn');
    const logoutBtn = document.getElementById('logoutBtn');
    const adminPassword = document.getElementById('adminPassword');
    const addProd = document.getElementById('addProd');
    const productsList = document.getElementById('productsList');
    const productGrid = document.getElementById('productGrid');

    document.getElementById('year').textContent = new Date().getFullYear();

    adminBtn.addEventListener('click', ()=> adminModal.classList.add('open'));
    closeAdmin.addEventListener('click', ()=> adminModal.classList.remove('open'));

    function loadProducts(){
      const raw = localStorage.getItem(productKey);
      if(!raw) return [];
      try{ return JSON.parse(raw) }catch(e){ return [] }
    }
    function saveProducts(list){ localStorage.setItem(productKey, JSON.stringify(list)); renderProducts() }

    function renderProducts(){
      const products = loadProducts();
      productGrid.innerHTML = '';
      products.forEach((p, i)=>{
        const el = document.createElement('div'); el.className='card';
        el.innerHTML = `
          <img src="${p.image}" alt="${p.title}">
          <h3>${p.title}</h3>
          <div class="tiny price">${p.price || ''}</div>
          <div class="actions">
            <a class="btn" href="${p.link || '#'}" target="_blank">Buy on Redbubble</a>
          </div>
        `;
        productGrid.appendChild(el);
      })

      // render admin list
      if(editor.style.display !== 'none'){
        productsList.innerHTML = '';
        products.forEach((p, i)=>{
          const row = document.createElement('div'); row.className='prod-item';
          row.innerHTML = `
            <img src="${p.image}">
            <div class="meta"><strong>${p.title}</strong><div class="tiny">${p.price || ''}</div><div class="tiny">${p.link || ''}</div></div>
            <div style="display:flex;gap:8px">
              <button class="btn" data-edit="${i}">Edit</button>
              <button class="btn" data-del="${i}">Delete</button>
            </div>
          `;
          productsList.appendChild(row);
        })
      }
    }

    // login flow
    function isLoggedIn(){ return sessionStorage.getItem('cd_admin')==='1' }
    function setLogged(on){ if(on) sessionStorage.setItem('cd_admin','1'); else sessionStorage.removeItem('cd_admin') }

    loginBtn.addEventListener('click', ()=>{
      const val = adminPassword.value.trim();
      if(!val) return alert('Enter password');
      if(val === DEFAULT_ADMIN_PASSWORD){
        setLogged(true);
        loginBox.style.display='none'; editor.style.display='block';
        renderProducts(); adminPassword.value='';
      } else alert('Wrong password');
    })

    logoutBtn.addEventListener('click', ()=>{
      setLogged(false); editor.style.display='none'; loginBox.style.display='block';
    })

    addProd.addEventListener('click', ()=>{
      const title = document.getElementById('pTitle').value.trim();
      const price = document.getElementById('pPrice').value.trim();
      const image = document.getElementById('pImage').value.trim() || 'https://images.unsplash.com/photo-1541099649105-f69ad21f3246?q=80&w=1400&auto=format&fit=crop&ixlib=rb-4.0.3&s=0e9f6e2f5d2d2b8b8a0f0e4f0b8d7a13';
      const link = document.getElementById('pLink').value.trim() || '#';
      if(!title){ return alert('Add a title') }
      const arr = loadProducts(); arr.unshift({title,price,image,link}); saveProducts(arr);
      document.getElementById('pTitle').value=''; document.getElementById('pPrice').value=''; document.getElementById('pImage').value=''; document.getElementById('pLink').value='';
    })

    // handle edit/delete via event delegation
    productsList.addEventListener('click', (e)=>{
      const edit = e.target.getAttribute('data-edit');
      const del = e.target.getAttribute('data-del');
      if(edit){
        const idx = Number(edit); const arr = loadProducts(); const p = arr[idx];
        const newTitle = prompt('Edit title', p.title); if(newTitle===null) return;
        const newPrice = prompt('Edit price', p.price); if(newPrice===null) return;
        const newImage = prompt('Edit image URL', p.image); if(newImage===null) return;
        const newLink = prompt('Edit Redbubble link', p.link); if(newLink===null) return;
        arr[idx] = {title:newTitle,price:newPrice,image:newImage,link:newLink}; saveProducts(arr);
      }
      if(del){
        if(!confirm('Delete this product?')) return;
        const idx = Number(del); const arr = loadProducts(); arr.splice(idx,1); saveProducts(arr);
      }
    })

    // auto-login state
    if(isLoggedIn()){
      loginBox.style.display='none'; editor.style.display='block';
    }

    // initial demo products if none
    if(!localStorage.getItem(productKey)){
      const demo = [
        {title:'Minimal Logo Hoodie', price:'KSh 3,200', image:'https://images.unsplash.com/photo-1512436991641-6745cdb1723f?q=80&w=1400&auto=format&fit=crop&ixlib=rb-4.0.3&s=6d55f8e4a3c4f7b6e2b2c9c5c0e4b7b2', link:'#'},
        {title:'Coco & Drift Tee', price:'KSh 2,100', image:'https://images.unsplash.com/photo-1541099649105-f69ad21f3246?q=80&w=1400&auto=format&fit=crop&ixlib=rb-4.0.3&s=0e9f6e2f5d2d2b8b8a0f0e4f0b8d7a13', link:'#'}
      ];
      localStorage.setItem(productKey, JSON.stringify(demo));
    }

    renderProducts();

    // close modal on ESC
    window.addEventListener('keydown', (e)=>{ if(e.key==='Escape') adminModal.classList.remove('open') })
  </script>
